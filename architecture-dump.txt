/* src/arch/i386/boot32.S
 *
 * 32-bit protected-mode entry for RockOS (V86 / older PCs).
 * 64-bit path in src/arch/x86_64/boot.S is untouched.
 *
 * Contract with Limine/Multiboot1 (same as 64-bit):
 *   EAX = magic, EBX = info pointer on entry.
 *   kernel_main(magic, info) with both zero-extended to 32-bit.
 *
 * Paging: identity-map the first 1 GiB with 4 MiB large pages in a
 * single page directory (256 entries), then enable PG+PE. This matches
 * PMM_IDENTITY_MAP_LIMIT and needs no 64-bit tables.
 */
.code32

.global _start
.extern kernel_main

.set ALIGN,    1<<0
.set MEMINFO,  1<<1
.set VIDEO,    1<<2
.set FLAGS,    ALIGN | MEMINFO | VIDEO
.set MAGIC,    0x1BADB002
.set CHECKSUM, -(MAGIC + FLAGS)

.section .multiboot, "a"
.align 4
.long MAGIC
.long FLAGS
.long CHECKSUM
/* Keep the optional address-field slots so Multiboot graphics fields land at
 * their specified fixed offsets even when flag 16 is clear. */
.long 0
.long 0
.long 0
.long 0
.long 0
/* Linear framebuffer mode, 1024x768x32. */
.long 0
.long 1024
.long 768
.long 32

.section .bss
.align 4096
pd_table:
    .skip 4096
.global boot_stack_bottom
boot_stack_bottom:
    .skip 16384
.global boot_stack_top
boot_stack_top:

.section .text
_start:
    movl %eax, %esi        /* magic -> ESI (callee-saved across setup) */
    movl %ebx, %edi        /* info  -> EDI */
    mov $boot_stack_top, %esp

    /* Build PD: 256 x 4MiB large pages covering 0..1GiB, P+RW. */
    mov $pd_table, %eax
    mov $0, %ecx
.map_pd:
    mov %ecx, %edx
    shl $22, %edx          /* phys base = ecx * 4MiB */
    or $0x83, %edx         /* P | RW | PS(4MiB) */
    mov %edx, (%eax,%ecx,4)
    inc %ecx
    cmp $256, %ecx
    jne .map_pd

    /* Zero the remaining 768 entries (3..4GiB unmapped for now). */
    mov $256, %ecx
.clear_pd:
    movl $0, (%eax,%ecx,4)
    inc %ecx
    cmp $1024, %ecx
    jne .clear_pd

    mov $pd_table, %eax
    mov %eax, %cr3

    mov %cr4, %eax
    or $(1 << 4), %eax    /* PSE: allow 4MiB pages */
    mov %eax, %cr4

    mov %cr0, %eax
    or $0x80000001, %eax  /* PG + PE */
    mov %eax, %cr0

    /* Stack is identity-mapped, flat segments from BIOS are fine.
     * GDT/IDT are installed later by kernel_main via gdt32/idt32. */
    subl $8, %esp         /* keep ESP 16-byte aligned at the call (i386 ABI) */
    pushl %edi            /* info  */
    pushl %esi            /* magic */
    call kernel_main
    addl $16, %esp

.hang:
    cli
    hlt
    jmp .hang

.section .note.GNU-stack,"",@progbits

/* src/arch/i386/context32_switch.S
 *
 * 32-bit context switch. Mirrors src/arch/x86_64/context_switch.S.
 *
 * C prototype:
 *   void context_switch(uintptr_t* old_sp, uintptr_t new_sp);
 * On i386: old_sp at 4(%esp), new_sp at 8(%esp).
 * Callee-saved: ebx, ebp, esi, edi.
 */
.code32

.section .text
.global context_switch
.type context_switch, @function
context_switch:
    pushl %ebx
    pushl %ebp
    pushl %esi
    pushl %edi

    movl 20(%esp), %eax      /* old_sp (4 regs pushed + ret addr = 20) */
    movl %esp, (%eax)
    movl 24(%esp), %eax      /* new_sp */
    movl %eax, %esp

    popl %edi
    popl %esi
    popl %ebp
    popl %ebx
    ret

.size context_switch, .-context_switch
.section .note.GNU-stack,"",@progbits

/*
 * src/arch/i386/divdi3.c
 *
 * 64-bit division/modulo helpers for the 32-bit (i386) RockOS build.
 *
 * On i386 GCC lowers every uint64_t '/' and '%' to calls to __udivdi3,
 * __umoddi3, __divdi3 and __moddi3, normally supplied by libgcc. The
 * x86_64-elf toolchain has no 32-bit libgcc, so we provide them here.
 *
 * IMPORTANT: this file must never use '/' or '%' on 64-bit values itself
 * (that would recurse). Only shifts, compares and adds on 64-bit values,
 * plus native 32-bit divides.
 */

#include <stdbool.h>
#include <stdint.h>

/* edx:eax / divisor -> eax quotient, edx remainder. Requires hi < divisor. */
static inline uint32_t div_64_by_32(uint32_t hi, uint32_t lo, uint32_t divisor,
    uint32_t* remainder)
{
    uint32_t quotient, rem;
    __asm__ ("divl %4"
        : "=a"(quotient), "=d"(rem)
        : "a"(lo), "d"(hi), "rm"(divisor)
        : "cc");
    *remainder = rem;
    return quotient;
}

uint64_t __udivmoddi4(uint64_t n, uint64_t d, uint64_t* rem)
{
    if (d == 0) {
        /* Real hardware would raise #DE; keep going with a sentinel. */
        if (rem) *rem = n;
        return ~(uint64_t)0;
    }

    /* Fast path: divisor fits in 32 bits (ticks / 100, value % 10, ...). */
    if ((d >> 32) == 0) {
        uint32_t d32 = (uint32_t)d;
        uint32_t hi = (uint32_t)(n >> 32);
        uint32_t lo = (uint32_t)n;
        uint32_t q_hi = hi / d32;          /* native 32-bit divide */
        uint32_t r_hi = hi % d32;
        uint32_t r_lo;
        uint32_t q_lo = div_64_by_32(r_hi, lo, d32, &r_lo);
        if (rem) *rem = r_lo;
        return ((uint64_t)q_hi << 32) | q_lo;
    }

    /* Slow path: divisor >= 2^32, so the quotient is < 2^32. Shift-subtract. */
    {
        uint64_t q = 0;
        uint64_t r = 0;
        for (int i = 0; i < 64; i++) {
            uint64_t carry = r >> 63;
            r = (r << 1) | (n >> 63);
            n <<= 1;
            q <<= 1;
            if (carry || r >= d) {
                r -= d;
                q |= 1;
            }
        }
        if (rem) *rem = r;
        return q;
    }
}

__attribute__((used)) uint64_t __udivdi3(uint64_t n, uint64_t d)
{
    return __udivmoddi4(n, d, 0);
}

__attribute__((used)) uint64_t __umoddi3(uint64_t n, uint64_t d)
{
    uint64_t r;
    __udivmoddi4(n, d, &r);
    return r;
}

__attribute__((used)) int64_t __divdi3(int64_t a, int64_t b)
{
    bool negative = (a < 0) != (b < 0);
    uint64_t ua = a < 0 ? (uint64_t)0 - (uint64_t)a : (uint64_t)a;
    uint64_t ub = b < 0 ? (uint64_t)0 - (uint64_t)b : (uint64_t)b;
    uint64_t q = __udivmoddi4(ua, ub, 0);
    return negative ? (int64_t)((uint64_t)0 - q) : (int64_t)q;
}

__attribute__((used)) int64_t __moddi3(int64_t a, int64_t b)
{
    uint64_t ua = a < 0 ? (uint64_t)0 - (uint64_t)a : (uint64_t)a;
    uint64_t ub = b < 0 ? (uint64_t)0 - (uint64_t)b : (uint64_t)b;
    uint64_t r;
    __udivmoddi4(ua, ub, &r);
    return a < 0 ? (int64_t)((uint64_t)0 - r) : (int64_t)r;
}

/* src/arch/i386/gdt32_load.S
 *
 * 32-bit protected-mode GDT reload (V86 / older PCs).
 * Mirrors src/arch/x86_64/gdt_load.S but with 32-bit segments:
 *   0x08 = kernel code, 0x10 = kernel data, 0x28 = TSS.
 *
 * C prototype: void gdt_load(const gdt_pointer_t* pointer);
 * On i386 the pointer arrives on the stack at 4(%esp).
 */
.code32

.section .text
.global gdt_load
.type gdt_load, @function
gdt_load:
    movl 4(%esp), %eax
    lgdt (%eax)

    ljmp $0x08, $.Lgdt32_reloaded
.Lgdt32_reloaded:
    movw $0x10, %ax
    movw %ax, %ds
    movw %ax, %es
    movw %ax, %fs
    movw %ax, %gs
    movw %ax, %ss

    movw $0x28, %ax
    ltr %ax
    ret

.size gdt_load, .-gdt_load
.section .note.GNU-stack,"",@progbits

 * PIC IRQs), plus the dedicated Ring 3 syscall vector 128. Mirrors
 * src/arch/x86_64/isr.S.
 *
 * Frame layout (lowest address first) must match interrupt_frame_t32 in
 * idt.c when ROCKOS_ARCH_32 is defined:
 *   edi esi ebp esp_dummy ebx edx ecx eax
 *   vector error_code
 *   eip cs eflags esp ss          <- pushed by the CPU
 *
 * (esp_dummy is pushed by pusha; the CPU-pushed esp is the one after
 * eflags. The C struct reads the CPU frame, not the pusha slot.)
 */

.extern interrupt_dispatch
.extern task_interrupt_exit

.global ring3_syscall_stub
ring3_syscall_stub:
    pushl $0
    pushl $128
    jmp isr32_common

isr32_stub_\num:

    jmp isr32_common
    .section .rodata

/*
 * src/arch/i386/paging32.c
 *
 * 32-bit (non-PAE) paging for the i386 RockOS build. Same public API as
 * src/kernel/paging.c (x86-64, 4-level), so the rest of the kernel does not
 * care which one it links.
 *
 * Layout of every address space:
 *   0x00000000 - 0x3FFFFFFF  identity map, supervisor, 4 MiB large pages
 *                            (needs CR4.PSE, enabled in boot32.S)
 *   0x40000000 - 0xBFFFFFFF  free for user mappings (4 KiB pages)
 *   0xE0000000 - 0xFFFFFFFF  kernel MMIO window (framebuffer, PCI BARs);
 *                            page tables shared with the kernel directory
 *
 * The "pml4_address" parameters in paging.h are simply the page-directory
 * physical address here.
 */

#include "paging.h"

#include "pmm.h"

#include <stddef.h>
#include <stdint.h>

#define P32_PRESENT      0x001U
#define P32_WRITABLE     0x002U
#define P32_USER         0x004U
#define P32_PWT          0x008U
#define P32_PCD          0x010U
#define P32_LARGE        0x080U
#define P32_ADDR_MASK    0xFFFFF000U
#define P32_LARGE_FLAGS  (P32_PRESENT | P32_WRITABLE | P32_LARGE)
#define P32_TABLE_FLAGS  (P32_PRESENT | P32_WRITABLE)

#define P32_ENTRIES         1024U
#define P32_KERNEL_LARGE    256U            /* 256 x 4 MiB = 1 GiB */
#define P32_USER_LIMIT      0xC0000000U
#define P32_MMIO_BASE       0xE0000000U
#define P32_MMIO_FIRST_PDE  (P32_MMIO_BASE >> 22)

static uintptr_t kernel_address_space;
static uint32_t kernel_mmio_next_virtual = P32_MMIO_BASE;

static void clear_table(uint32_t* table) {
    for (size_t index = 0; index < P32_ENTRIES; index++) {
        table[index] = 0;
    }
}

static bool table_is_empty(const uint32_t* table) {
    for (size_t index = 0; index < P32_ENTRIES; index++) {
        if (table[index] & P32_PRESENT) return false;
    }
    return true;
}

static bool valid_page_address(uintptr_t address) {
    return (address & (PMM_PAGE_SIZE - 1)) == 0;
}

static inline void invalidate_page(uintptr_t address) {
    asm volatile("invlpg (%0)" : : "r"(address) : "memory");
}

uintptr_t paging_current_address_space(void) {
    uintptr_t address;
    asm volatile("mov %%cr3, %0" : "=r"(address));
    return address;
}

void paging_init(void) {
    if (!kernel_address_space) {
        kernel_address_space = paging_current_address_space() & P32_ADDR_MASK;
    }
}

uintptr_t paging_address_space_create(void) {
    uint32_t* pd = (uint32_t*)pmm_alloc(1);
    if (!pd) return 0;

    paging_init();
    clear_table(pd);
    for (uint32_t index = 0; index < P32_KERNEL_LARGE; index++) {
        pd[index] = (index << 22) | P32_LARGE_FLAGS;
    }

    /* Share the kernel MMIO window (framebuffer etc.) with every address
     * space so drawing keeps working if a task with a user directory is
     * running when the scheduler switches to the UI loop. */
    if (kernel_address_space) {
        const uint32_t* kernel_pd = (const uint32_t*)kernel_address_space;
        for (uint32_t index = P32_MMIO_FIRST_PDE; index < P32_ENTRIES; index++) {
            pd[index] = kernel_pd[index];
        }
    }
    return (uintptr_t)pd;
}

bool paging_map_user_page(uintptr_t pd_address, uintptr_t virtual_address,
    uintptr_t physical_address, bool writable) {
    if (!pd_address || !valid_page_address(pd_address) ||
        !valid_page_address(virtual_address) ||
        !valid_page_address(physical_address) ||
        physical_address >= PMM_IDENTITY_MAP_LIMIT ||
        virtual_address >= P32_USER_LIMIT) {
        return false;
    }

    uint32_t* pd = (uint32_t*)pd_address;
    size_t pd_index = virtual_address >> 22;
    size_t pt_index = (virtual_address >> 12) & 0x3FF;
    uint32_t* pt;
    bool allocated_pt = false;

    if (!(pd[pd_index] & P32_PRESENT)) {
        pt = (uint32_t*)pmm_alloc(1);
        if (!pt) return false;
        clear_table(pt);
        pd[pd_index] = (uint32_t)(uintptr_t)pt |
            P32_PRESENT | P32_WRITABLE | P32_USER;
        allocated_pt = true;
    } else {
        if (pd[pd_index] & P32_LARGE) return false;
        pt = (uint32_t*)(uintptr_t)(pd[pd_index] & P32_ADDR_MASK);
    }

    if (pt[pt_index] & P32_PRESENT) {
        if (allocated_pt) {
            pd[pd_index] = 0;
            pmm_free(pt, 1);
        }
        return false;
    }

    pd[pd_index] |= P32_USER | P32_WRITABLE;
    pt[pt_index] = (uint32_t)physical_address | P32_PRESENT | P32_USER |
        (writable ? P32_WRITABLE : 0U);
    return true;
}

bool paging_unmap_user_page(uintptr_t pd_address, uintptr_t virtual_address,
    uintptr_t* physical_address) {
    if (!pd_address || !valid_page_address(pd_address) ||
        !valid_page_address(virtual_address) ||
        virtual_address >= P32_USER_LIMIT || !physical_address) {
        return false;
    }

    uint32_t* pd = (uint32_t*)pd_address;
    size_t pd_index = virtual_address >> 22;
    size_t pt_index = (virtual_address >> 12) & 0x3FF;
    if (!(pd[pd_index] & P32_PRESENT) || (pd[pd_index] & P32_LARGE)) {
        return false;
    }

    uint32_t* pt = (uint32_t*)(uintptr_t)(pd[pd_index] & P32_ADDR_MASK);
    uint32_t entry = pt[pt_index];
    if (!(entry & P32_PRESENT) || !(entry & P32_USER)) {
        return false;
    }

    *physical_address = (uintptr_t)(entry & P32_ADDR_MASK);
    pt[pt_index] = 0;
    invalidate_page(virtual_address);
    if (table_is_empty(pt)) {
        pd[pd_index] = 0;
        if (!pmm_free(pt, 1)) return false;
    }
    return true;
}

bool paging_address_space_destroy(uintptr_t pd_address) {
    if (!pd_address || (pd_address & (PMM_PAGE_SIZE - 1)) != 0) {
        return false;
    }

    uint32_t* pd = (uint32_t*)pd_address;
    if ((pd[0] & P32_LARGE_FLAGS) != P32_LARGE_FLAGS) return false;
    /* User page tables are freed by paging_unmap_user_page; the shared
     * kernel MMIO tables must NOT be freed here. */
    return pmm_free(pd, 1);
}

void paging_load_address_space(uintptr_t pd_address) {
    if (!pd_address || (pd_address & (PMM_PAGE_SIZE - 1)) != 0) return;
    asm volatile("mov %0, %%cr3" : : "r"(pd_address) : "memory");
}

bool paging_map_kernel_mmio(uintptr_t physical_address, size_t byte_count,
    uintptr_t* virtual_address) {
    if (!physical_address || !byte_count || !virtual_address) return false;

    uint32_t page_offset = (uint32_t)(physical_address & (PMM_PAGE_SIZE - 1));
    uint32_t physical_page = (uint32_t)physical_address & P32_ADDR_MASK;
    uint64_t span = (uint64_t)page_offset + byte_count;
    uint64_t page_count = (span + (PMM_PAGE_SIZE - 1)) >> 12;
    uint64_t map_size = page_count << 12;

    if ((uint64_t)physical_page + map_size > 0x100000000ULL) return false;
    if ((uint64_t)kernel_mmio_next_virtual + map_size > 0x100000000ULL) {
        return false;
    }

    paging_init();
    uint32_t* pd = (uint32_t*)kernel_address_space;
    uint32_t map_start = kernel_mmio_next_virtual;

    for (uint64_t page = 0; page < page_count; page++) {
        uint32_t va = map_start + (uint32_t)(page << 12);
        size_t pd_index = va >> 22;
        size_t pt_index = (va >> 12) & 0x3FF;
        uint32_t* pt;

        if (!(pd[pd_index] & P32_PRESENT)) {
            pt = (uint32_t*)pmm_alloc(1);
            if (!pt) return false;
            clear_table(pt);
            pd[pd_index] = (uint32_t)(uintptr_t)pt | P32_TABLE_FLAGS;
        } else {
            if (pd[pd_index] & P32_LARGE) return false;
            pt = (uint32_t*)(uintptr_t)(pd[pd_index] & P32_ADDR_MASK);
        }
        if (pt[pt_index] & P32_PRESENT) return false;

        /* Uncached (PWT|PCD), same as the x86-64 mapping. */
        pt[pt_index] = (physical_page + (uint32_t)(page << 12)) |
            P32_PRESENT | P32_WRITABLE | P32_PWT | P32_PCD;
        invalidate_page(va);
    }

    kernel_mmio_next_virtual += (uint32_t)map_size;
    *virtual_address = (uintptr_t)(map_start + page_offset);
    return true;
}

uintptr_t paging_kernel_address_space(void) {
    paging_init();
    return kernel_address_space;
}

void paging_restore_kernel_address_space(void) {
    paging_load_address_space(paging_kernel_address_space());
}

/*
 * src/arch/x86_64/isr.S
 *
 * Stubs for vectors 0-47 (CPU exceptions + remapped PIC IRQs), plus the
 * dedicated Ring 3 syscall vector 128. Every stub reaches isr_common, which
 * builds one uniform frame and calls interrupt_dispatch(frame) in idt.c.
 *
 * Frame layout (lowest address first) must match interrupt_frame_t in idt.c:
 *   r15 r14 r13 r12 r11 r10 r9 r8 rbp rdi rsi rdx rcx rbx rax
 *   vector error_code
 *   rip cs rflags rsp ss          <- pushed by the CPU
 *
 * An interrupt can arrive with arbitrary RSP alignment. isr_common aligns a
 * temporary stack for its C call and restores the original frame pointer.
 */

.code64

    cld                         /* SysV ABI: DF must be clear when calling C */
    movq %rsp, %rdi             /* arg0 = pointer to frame */
    movq %rsp, %rax             /* preserve frame pointer across the C call */
    andq $-16, %rsp              /* SysV ABI: align RSP before CALL */
    subq $16, %rsp
    movq %rax, 8(%rsp)
    call interrupt_dispatch
    movq 8(%rsp), %rsp          /* restore the interrupt frame stack */

    cmpq $128, 120(%rsp)        /* syscalls can return to user mode */
    je .Lisr_skip_task_exit

    /* Schedule only after dispatch has completed and the PIC EOI was sent. */
    movq %rsp, %rax
    andq $-16, %rsp
    subq $16, %rsp
    movq %rax, 8(%rsp)
    call task_interrupt_exit
    movq 8(%rsp), %rsp          /* restore this task's interrupt frame */
